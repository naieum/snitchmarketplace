# Security headers tuned for the 10 supported ad platforms.
# Format-agnostic: lib/apply_headers.sh adapts to the user's stack
# (next.config.js headers(), _headers for Pages, vercel.json, .htaccess, nginx.conf, etc.).
#
# Replace {{NONCE}} with a per-request nonce generator (Edge middleware in Next, Workers, Pages Functions).
# 'unsafe-inline' for script-src is intentionally absent — pixels must use a nonce or hashes.

# ----------------------------------------------------------------------
# Strict-Transport-Security
# Force HTTPS for 1 year, include subdomains, and submit to the HSTS preload list.
# ----------------------------------------------------------------------
Strict-Transport-Security: max-age=31536000; includeSubDomains; preload

# ----------------------------------------------------------------------
# Content-Security-Policy
# Allowlists every script-loader, iframe target, and image source the supported
# ad platforms use. Conservative — start here and remove platforms you don't run.
# ----------------------------------------------------------------------
Content-Security-Policy: default-src 'self'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'; upgrade-insecure-requests; script-src 'self' 'nonce-{{NONCE}}' https://www.googletagmanager.com https://www.google-analytics.com https://googleads.g.doubleclick.net https://www.googleadservices.com https://www.gstatic.com https://connect.facebook.net https://bat.bing.com https://snap.licdn.com https://*.licdn.com https://analytics.tiktok.com https://*.tiktok.com https://static.ads-twitter.com https://analytics.twitter.com https://s.pinimg.com https://ct.pinterest.com https://*.redditstatic.com https://www.redditstatic.com https://sc-static.net https://tr.snapchat.com https://*.snapchat.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; img-src 'self' data: blob: https://www.google-analytics.com https://www.googletagmanager.com https://www.google.com https://googleads.g.doubleclick.net https://www.facebook.com https://*.facebook.com https://bat.bing.com https://*.linkedin.com https://analytics.tiktok.com https://t.co https://*.twimg.com https://ct.pinterest.com https://*.pinimg.com https://*.redditstatic.com https://tr.snapchat.com https://*.snapchat.com; # ----------------------------------------------------------------------
# X-Content-Type-Options
# Prevent MIME-type sniffing — pixels often serve as image/gif; sniffers can mis-execute.
# ----------------------------------------------------------------------
X-Content-Type-Options: nosniff

# ----------------------------------------------------------------------
# X-Frame-Options + frame-ancestors (CSP) — clickjacking defense.
# DENY blocks all framing. If you need partner embeds, use SAMEORIGIN or a CSP allowlist.
# ----------------------------------------------------------------------
X-Frame-Options: DENY

# ----------------------------------------------------------------------
# Referrer-Policy
# strict-origin-when-cross-origin balances analytics attribution (referrer present
# for same-origin) with cross-origin privacy (origin only).
# ----------------------------------------------------------------------
Referrer-Policy: strict-origin-when-cross-origin

# ----------------------------------------------------------------------
# Permissions-Policy
# Minimal. Add features as your pages need them.
# ----------------------------------------------------------------------
Permissions-Policy: geolocation=(), microphone=(), camera=(), payment=(self), usb=(), interest-cohort=()

# ----------------------------------------------------------------------
# Cross-Origin-* (defense-in-depth)
# ----------------------------------------------------------------------
Cross-Origin-Opener-Policy: same-origin
Cross-Origin-Resource-Policy: same-site

# ----------------------------------------------------------------------
# Cache-Control (apply to HTML responses, NOT static assets)
# Pixels mutate often; don't cache the page that loads them.
# ----------------------------------------------------------------------
Cache-Control: private, no-cache, max-age=0, must-revalidate

# ----------------------------------------------------------------------
# Notes:
# - Validate with https://securityheaders.com (target A) and https://observatory.mozilla.org (target 90+).
# - Monitor browser console for `Refused to load…` CSP errors; whitelist the missing domain narrowly.
# - Some ad platforms (notably Meta) still rely on `'unsafe-eval'` in their pixel; avoid where possible.
#   If unavoidable, gate behind a per-page CSP override rather than the global default.
# - Consent Mode v2 sometimes requires connect-src to your CMP vendor's domain
#   (e.g., onetrust.com, cookiebot.com, klaro.org). Add per the CMP you pick.
